use std::{fs::{self,OpenOptions},io::Write,thread,time::Duration};
fn main(){
 writeln!(OpenOptions::new().append(true).create(true).open(std::env::var_os("TBS_BODY").unwrap()).unwrap(),"{}",std::process::id()).unwrap();
 let release=std::env::var_os("TBS_RELEASE").unwrap();
 while !std::path::Path::new(&release).exists(){thread::sleep(Duration::from_millis(10));}
 fs::metadata(release).unwrap();
}
